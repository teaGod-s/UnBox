# M5.1 JS 爬虫（goja 运行时 + 规则引擎 + Spider Provider）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 用 goja 在桌面端运行 TVBox 爬虫 JS，让 `.js` 爬虫站点（drpy2 声明式 rule 或自定义函数）可浏览、搜索、详情、播放。

**Tech Stack:** `github.com/dop251/goja`（纯 Go JS 引擎）、`github.com/PuerkitoBio/goquery`（jQuery 式选择器，封装 x/net/html）。

**Spec:** `docs/superpowers/specs/2026-08-29-unbox-m5-js-engine-design.md`

## Global Constraints

- 模块路径 `github.com/unbox/unbox`，Go 1.26.3（mise 钉死）。
- 新增包 `internal/crawler` **不得 import** `internal/provider`、Wails；它只暴露自己的类型，映射交给 tvbox。

	"github.com/dop251/goja"

type Engine struct {
	vm *goja.Runtime
	hc *http.Client // Task 3 起用于 req()

func New() *Engine {

// Load 执行一段爬虫 JS（声明式 rule 或函数定义），失败返回错误。
func (e *Engine) Load(src string) error {
	_, err := e.vm.RunString(src)

func (e *Engine) Call(name string, args ...goja.Value) (goja.Value, error) {
	fn, ok := goja.AssertFunction(e.vm.Get(name))
	if !ok {
		return nil, fmt.Errorf("爬虫未定义函数 %s", name)
	}
	return fn(goja.Undefined(), args...)

```go
package crawler

import (
	"fmt"
	"io"
	"net/http"
	"strings"
	"time"
)

// reqOpts 是 req 第二个参数（JS 对象）映射的 Go 结构。
type reqOpts struct {
	Method  string            `json:"method"`
	Headers map[string]string `json:"headers"`
	Data    string            `json:"data"`
	Timeout int64             `json:"timeout"`
}

// installReq 把 req(url, opts) 注入 vm。opts.data 为字符串时作 body；
// 有 method 则用其，否则 GET。
func (e *Engine) installReq(hc *http.Client) {
	_ = e.vm.Set("req", func(call goja.FunctionCall) goja.Value {
		url := call.Argument(0).String()
		var o reqOpts
		if !goja.IsUndefined(call.Argument(1)) {
			_ = e.vm.ExportTo(call.Argument(1), &o)
		}
		method := o.Method
		if method == "" {
			method = http.MethodGet
		}
		var body io.Reader
		if o.Data != "" {
			body = strings.NewReader(o.Data)
		}
		req, err := http.NewRequest(method, url, body)
		if err != nil {
			panic(e.vm.NewGoError(fmt.Errorf("req 构造请求失败: %w", err)))
		}
		for k, v := range o.Headers {
			req.Header.Set(k, v)
		}
		if req.Header.Get("User-Agent") == "" {
			req.Header.Set("User-Agent", "okhttp/3.12.11")
		}
		resp, err := hc.Do(req)
		if err != nil {
			panic(e.vm.NewGoError(fmt.Errorf("req 请求失败: %w", err)))
		}
		defer resp.Body.Close()
		b, err := io.ReadAll(io.LimitReader(resp.Body, 16<<20))
		if err != nil {
			panic(e.vm.NewGoError(fmt.Errorf("req 读响应失败: %w", err)))
		}
		out := e.vm.NewObject()
		_ = out.Set("content", string(b))
		_ = out.Set("statusCode", resp.StatusCode)
		_ = out.Set("finalUrl", resp.Request.URL.String())
		return out
	})
}
```

```go
package crawler

import (
	"regexp"
	"strings"

	"github.com/PuerkitoBio/goquery"
	"github.com/dop251/goja"
)

// evalRule 执行一条 pdfh/pdfa 规则，返回全部命中值。
// rule 形如 "ul&&a&&href" 或 "li&&Text()"。
func evalRule(doc *goquery.Document, rule string) []string {
	sel := doc.Selection
	for _, seg := range strings.Split(rule, "&&") {
		seg = strings.TrimSpace(seg)
		switch {
		case seg == "Text()":
			sel = sel.Map(func(_ int, s *goquery.Selection) string { return strings.TrimSpace(s.Text()) })
		case seg == "Html()":
			sel = sel.Map(func(_ int, s *goquery.Selection) string {
				h, _ := goquery.OuterHtml(s)
				return h
			})
		case seg == "href":
			sel = sel.Map(func(_ int, s *goquery.Selection) string { v, _ := s.Attr("href"); return v })
		case seg == "src":
			sel = sel.Map(func(_ int, s *goquery.Selection) string { v, _ := s.Attr("src"); return v })
		case strings.HasPrefix(seg, "attr("):
			name := strings.TrimSuffix(strings.TrimPrefix(seg, "attr("), ")")
			name = strings.Trim(name, `'"`)
			sel = sel.Map(func(_ int, s *goquery.Selection) string { v, _ := s.Attr(name); return v })
		case strings.HasPrefix(seg, "match("):
			re := regexp.MustCompile(strings.TrimSuffix(strings.TrimPrefix(seg, "match("), ")"))
			sel = sel.Map(func(_ int, s *goquery.Selection) string { return re.FindString(s.Text()) })
		case strings.HasPrefix(seg, "split("):
			sep := strings.TrimSuffix(strings.TrimPrefix(seg, "split("), ")")
			sel = sel.Map(func(_ int, s *goquery.Selection) string { return strings.Split(s.Text(), sep)[0] })
		default: // CSS 选择器
			sel = sel.Find(seg)
		}
	}
	out := []string{}
	sel.Each(func(_ int, s *goquery.Selection) { out = append(out, s.Text()) })
	return out
}

func (e *Engine) installRule() {
	doc := func(html string) *goquery.Document {
		d, _ := goquery.NewDocumentFromReader(strings.NewReader(html))
		return d
	}
	_ = e.vm.Set("pdfh", func(call goja.FunctionCall) goja.Value {
		res := evalRule(doc(call.Argument(0).String()), call.Argument(1).String())
		if len(res) == 0 {
			return goja.Undefined()
		}
		return e.vm.ToValue(res[0])
	})
	_ = e.vm.Set("pdfa", func(call goja.FunctionCall) goja.Value {
		return e.vm.ToValue(evalRule(doc(call.Argument(0).String()), call.Argument(1).String()))
	})
	_ = e.vm.Set("pd", func(call goja.FunctionCall) goja.Value {
		res := evalRule(doc(call.Argument(0).String()), call.Argument(1).String())
		return e.vm.ToValue(strings.Join(res, call.Argument(2).String()))
	})
}
```

package crawler

import (
	"crypto/md5"
	"encoding/base64"
	"encoding/hex"
	"log"
	"time"
)

func (e *Engine) installHelpers() {
	_ = e.vm.Set("log", func(call goja.FunctionCall) goja.Value {
		args := make([]any, len(call.Arguments))
		for i, a := range call.Arguments {
			args[i] = a.Export()
		}
		log.Print(args...)
		return goja.Undefined()
	})
	_ = e.vm.Set("base64Encode", func(call goja.FunctionCall) goja.Value {
		return e.vm.ToValue(base64.StdEncoding.EncodeToString([]byte(call.Argument(0).String())))
	})
	_ = e.vm.Set("base64Decode", func(call goja.FunctionCall) goja.Value {
		b, _ := base64.StdEncoding.DecodeString(call.Argument(0).String())

		h := md5.Sum([]byte(call.Argument(0).String()))
		return e.vm.ToValue(hex.EncodeToString(h[:]))
	})
	_ = e.vm.Set("sleep", func(call goja.FunctionCall) goja.Value {
		time.Sleep(time.Duration(call.Argument(0).ToInteger()) * time.Millisecond)
		return goja.Undefined()
	})

package crawler

import (

	"fmt"

	"github.com/dop251/goja"

// ruleObj 是从 vm 里导出 rule 对象的包装。
type ruleObj struct {
	Host       string   `json:"host"`
	SearchURL  string   `json:"searchUrl"`
	DetailURL  string   `json:"detailUrl"`
	ClassName  string   `json:"class_name"`
	ClassURL   string   `json:"class_url"`
	VodSel     string   `json:"vod_selector"`
	NameSel    string   `json:"name_selector"`
	PicSel     string   `json:"pic_selector"`
	TypeSel    string   `json:"type_selector"`
	RemarkSel  string   `json:"remarks_selector"`
}

func (e *Engine) rule() (*ruleObj, error) {
	obj, ok := e.vm.Get("rule").(*goja.Object)
	if !ok {
		return nil, fmt.Errorf("爬虫未定义 rule 对象")
	}
	var r ruleObj
	if err := e.vm.ExportTo(obj, &r); err != nil {
		return nil, err
	}
	return &r, nil
}

// callList 调用爬虫的列表逻辑，返回 []Vod。当前实现：直接读 fixture 内置的
// getList() 函数；声明式 rule 的选择器解析在验收时用真实爬虫校准后补。
func (e *Engine) callList(fn string, args ...goja.Value) ([]Vod, error) {
	v, err := e.Call(fn, args...)
	if err != nil {
		return nil, err
	}
	var vods []Vod
	if err := e.vm.ExportTo(v, &vods); err != nil {
		return nil, fmt.Errorf("爬虫列表返回非数组: %w", err)
	}
	return vods, nil
}

func (e *Engine) LoadFromURL(ctx context.Context, u string) error {
	req, _ := http.NewRequestWithContext(ctx, http.MethodGet, u, nil)
	req.Header.Set("User-Agent", "okhttp/3.12.11")
	resp, err := e.hc.Do(req)
	if err != nil { return err }
	defer resp.Body.Close()
	b, err := io.ReadAll(io.LimitReader(resp.Body, 4<<20))
	if err != nil { return err }
	return e.Load(string(b))
}
